cmake_minimum_required(VERSION 3.16)
project(hjb_mm CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# no -ffast-math: serial and parallel runs must give bitwise-identical results
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
option(NATIVE "Tune for the build machine (-march=native)" ON)
if(NATIVE)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "NVHPC")
    add_compile_options(-tp=native)
  else()
    add_compile_options(-march=native)
  endif()
endif()

find_package(OpenMP REQUIRED)
find_package(MPI)

add_executable(hjb_serial src/main_serial.cpp)
target_link_libraries(hjb_serial PRIVATE OpenMP::OpenMP_CXX)

if(MPI_FOUND)
  add_executable(hjb_mpi src/main_mpi.cpp)
  target_link_libraries(hjb_mpi PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX)
endif()

# GPU build (MPI + OpenMP target offload), off by default:
#   cmake -S . -B build-gpu -DGPU=ON -DCMAKE_CXX_COMPILER=nvc++      (NVIDIA HPC SDK, recommended)
#   cmake -S . -B build-gpu -DGPU=ON -DCMAKE_CXX_COMPILER=clang++    (LLVM with NVPTX offload)
#   cmake -S . -B build-gpu -DGPU=ON                                 (GCC with gcc-offload-nvptx)
# GPU_ARCH: compute capability, 86 for the A40.
option(GPU "Build hjb_gpu (OpenMP target offload)" OFF)
set(GPU_ARCH 86 CACHE STRING "GPU compute capability")
if(GPU)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "NVHPC")
    set(GPU_FLAGS -mp=gpu -gpu=cc${GPU_ARCH} -Minfo=mp)
  elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(GPU_FLAGS -fopenmp -fopenmp-targets=nvptx64-nvidia-cuda --offload-arch=sm_${GPU_ARCH})
  else()
    set(GPU_FLAGS -fopenmp -foffload=nvptx-none -foffload-options=-lm -fcf-protection=none -fno-stack-protector)
  endif()
  add_executable(hjb_gpu src/main_gpu.cpp)
  target_compile_options(hjb_gpu PRIVATE ${GPU_FLAGS})
  target_link_options(hjb_gpu PRIVATE ${GPU_FLAGS})
  target_link_libraries(hjb_gpu PRIVATE MPI::MPI_CXX)
endif()
